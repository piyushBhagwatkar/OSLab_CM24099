Zombie Process 


#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>


int main() {
    pid_t pid = fork();


    if (pid > 0) {
        // Parent process
        printf("Parent Process ID: %d\n", getpid());
        sleep(10);  // Parent sleeps, child becomes zombie
        printf("Parent waking up\n");
    }
    else if (pid == 0) {
        // Child process
        printf("Child Process ID: %d\n", getpid());
        printf("Child exiting...\n");
        exit(0);  // Child exits immediately
    }
    else {
        printf("Fork failed\n");
    }


    return 0;
}


































Orphan  Process 


#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>


int main() {
    pid_t pid = fork();


    if (pid > 0) {
        // Parent process
        printf("Parent Process ID: %d\n", getpid());
        printf("Parent exiting...\n");
        exit(0);  // Parent exits immediately
    }
    else if (pid == 0) {
        // Child process
        sleep(5);  // Wait so parent exits first
        printf("Child Process ID: %d\n", getpid());
        printf("Child's New Parent ID (PPID): %d\n", getppid());
    }
    else {
        printf("Fork failed\n");
    }


    return 0;
}


































Process using Fork () 


#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>


int main() {
    pid_t pid;


    pid = fork();  // Create child process


    if (pid < 0) {
        printf("Fork failed!\n");
    }
    else if (pid == 0) {
        // Child Process
        printf("\n--- Child Process ---\n");
        printf("Child PID: %d\n", getpid());
        printf("Parent PID (PPID): %d\n", getppid());
    }
    else {
        // Parent Process
        printf("\n--- Parent Process ---\n");
        printf("Parent PID: %d\n", getpid());
        printf("Child PID: %d\n", pid);
    }


    return 0;
}
